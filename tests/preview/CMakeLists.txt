cmake_minimum_required(VERSION 3.16)
project(preview_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
add_executable(preview-tests preview_tests.cpp ../../ipc/preview_protocol.c)
target_include_directories(preview-tests PRIVATE ../../ipc/include)
target_compile_options(preview-tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME preview-protocol-and-lifetime COMMAND preview-tests)
add_executable(monitor-under-test ../../monitor/main.cpp)
target_compile_options(monitor-under-test PRIVATE -Wall -Wextra -Werror)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME monitor-ui-crash-isolation COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/monitor_recovery.py $<TARGET_FILE:monitor-under-test>)
